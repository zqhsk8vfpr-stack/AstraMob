local function fireSilentAim(aimPos)
    if not remote then return end
    pcall(function()
        remote:FireServer("MOUSE", aimPos)
    end)
end

local function prepareSilentAim()
    if not getgenv().SilentAimEnabled then return end
    local hitChance = getgenv().SilentAimHitChance or 85
    if math.random(100) > hitChance then return end

    local targetPlayer = SilentTarget
    if not targetPlayer or not targetPlayer.character then return end

    local targetChar = targetPlayer.character
    if getgenv().SilentAimKnockCheck and IsKnocked(targetChar) then return end
    if getgenv().SilentAimForcefieldCheck and hasForcefield(targetChar) then return end
    if getgenv().SilentAimFriendCheck and targetPlayer.player:IsFriendsWith(LocalPlayer.UserId) then return end

    local hum = targetChar:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end

    local aimPart, customPoint = getSilentAimPart(targetChar, targetPlayer)
    if not aimPart then return end

    if getgenv().SilentAimWallCheck and not isVisible(aimPart.Position, targetPlayer) then return end

    local aimPos = customPoint or computeSilentAimPosition(aimPart, targetPlayer,
        getgenv().SilentAimPrediction,
        getgenv().SilentAimAutoPrediction,
        getgenv().SilentAimHitChance or 85)

    local antiCurve = getgenv().SilentAimAntiCurve or 8
    if antiCurve > 0 then
        local dir = (aimPos - Camera.CFrame.Position).Unit
        local angle = math.deg(math.acos(Camera.CFrame.LookVector:Dot(dir)))
        if angle > antiCurve then return end
    end

    if getgenv().SilentAimAntiGround then
        local groundHit = GetGroundIntersection(aimPos, targetChar)
        if groundHit then return end
    end

    fireSilentAim(aimPos)
end

local function onToolActivated()
    prepareSilentAim()
end

local hookedTools = {}
local function hookTool(tool)
    if not tool:IsA("Tool") or hookedTools[tool] then return end
    hookedTools[tool] = true
    tool.Activated:Connect(onToolActivated)
end
local function hookAllTools()
    for _, t in ipairs(LocalPlayer.Backpack:GetChildren()) do hookTool(t) end
    if LocalPlayer.Character then
        for _, t in ipairs(LocalPlayer.Character:GetChildren()) do hookTool(t) end
    end
end
hookAllTools()
LocalPlayer.CharacterAdded:Connect(function(char)
    task.wait(0.5) hookAllTools()
    char.ChildAdded:Connect(function(c) task.wait(0.1) hookTool(c) end)
end)
LocalPlayer.Backpack.ChildAdded:Connect(function(c) task.wait(0.1) hookTool(c) end)

-- Main aimbot loop (polling-based for all modes to avoid UI event consumption)
local prevLookVectors = {}
local function recordLookVector()
    table.insert(prevLookVectors, Camera.CFrame.LookVector)
    if #prevLookVectors > 5 then table.remove(prevLookVectors, 1) end
end

local function getMovementDirection()
    if #prevLookVectors < 2 then return nil end
    local avg = Vector3.zero
    for i = 2, #prevLookVectors do
        avg = avg + (prevLookVectors[i] - prevLookVectors[i-1])
    end
    return avg.Unit
end

local function getEasedValue(t, style, dir)
    style = style or "Linear"
    dir = dir or "Out"
    if style == "Quad" and dir == "Out" then
        return 1 - (1 - t) * (1 - t)
    end
    return t
end

RunService.RenderStepped:Connect(function()
    recordLookVector()

    if not getgenv().Enabled then
        currentTarget = nil
        SilentTarget = nil
        return
    end

    local activation = getgenv().Activation or "Hold"
    local shouldLock = false
    local enemiesList = nil

    if activation == "Rage" then
        shouldLock = true
        enemiesList = getValidEnemies()
    elseif activation == "Mobile" then
        if isTouching then
            shouldLock = true
            enemiesList = getValidEnemies()
        else
            currentTarget = nil
            SilentTarget = nil
            return
        end
    elseif activation == "Hold" then
        local key = getKeyCode(getgenv().AimbotKey or "E")
        if UserInputService:IsKeyDown(key) then
            shouldLock = true
            enemiesList = getValidEnemies()
        else
            currentTarget = nil
            SilentTarget = nil
            return
        end
    elseif activation == "Toggle" then
        local key = getKeyCode(getgenv().AimbotKey or "E")
        local isDown = UserInputService:IsKeyDown(key)
        if isDown and not previousKeyState then
            camLockActive = not camLockActive
            if not camLockActive then
                currentTarget = nil
                SilentTarget = nil
            end
        end
        previousKeyState = isDown
        if camLockActive then
            shouldLock = true
            enemiesList = getValidEnemies()
        else
            currentTarget = nil
            SilentTarget = nil
            return
        end
    end

    if not shouldLock then
        currentTarget = nil
        SilentTarget = nil
        return
    end

    if not enemiesList or #enemiesList == 0 then
        currentTarget = nil
        SilentTarget = nil
        return
    end

    if currentTarget then
        local stillValid = false
        for _, data in ipairs(enemiesList) do
            if data.player == currentTarget.player then
                stillValid = true
                break
            end
        end
        if not stillValid then
            currentTarget = nil
            SilentTarget = nil
        end
    end

    if not currentTarget then
        local targetData = selectTarget(enemiesList)
        if targetData then
            currentTarget = targetData
        else
            currentTarget = enemiesList[1]
        end
        if not currentTarget then
            return
        end
    end

    if getgenv().SilentAimEnabled then
        SilentTarget = currentTarget
    else
        SilentTarget = nil
    end

    local char = currentTarget.character
    local root = currentTarget.rootPart

    local aimPart, customPoint = getAimPart(char, currentTarget.player)
    if not aimPart then
        aimPart = root
    end

    local aimPos = customPoint or aimPart.Position

    if getgenv().Prediction then
        local vel = aimPart.AssemblyLinearVelocity
        local hPred = getgenv().PredictionX or 0.115
        local vPred = getgenv().PredictionY or 0.115
        local zPred = getgenv().PredictionZ or 0.115
        aimPos = aimPos + Vector3.new(vel.X * hPred, vel.Y * vPred, vel.Z * zPred)
    end

    if getgenv().AutoPrediction then
        local hMult = getgenv().PredictionX or 0.115
        local vMult = getgenv().PredictionY or 0.115
        local zMult = getgenv().PredictionZ or 0.115
        local vel = aimPart.AssemblyLinearVelocity
        aimPos = aimPos + Vector3.new(vel.X * hMult, vel.Y * vMult, vel.Z * zMult)
    end

    local airborne = IsAirborne(char)
    if airborne then
        local jumpOff = getgenv().JumpOffset or 0
        if jumpOff ~= 0 then
            aimPos = aimPos + Vector3.new(0, jumpOff, 0)
        end
    else
        local fallOff = getgenv().FallOffset or 0
        if fallOff ~= 0 then
            aimPos = aimPos + Vector3.new(0, fallOff, 0)
        end
    end

    if getgenv().AimbotWallCheck and not isVisible(aimPos, currentTarget.player) then
        return
    end

    if getgenv().TargetOutline then
        local dirToTarget = (aimPos - Camera.CFrame.Position).Unit
        local angularError = math.deg(math.acos(Camera.CFrame.LookVector:Dot(dirToTarget)))
        local ray = Camera:ViewportPointToRay(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
        local rel = aimPos - ray.Origin
        local proj = rel:Dot(ray.Direction)
        local closestOnRay = ray.Origin + ray.Direction * proj
        local perpDist = (aimPos - closestOnRay).Magnitude
        local shouldAssist = true
        if perpDist > 8 then shouldAssist = false end
        if angularError < 2.0 then
            local desiredCF = CFrame.lookAt(Camera.CFrame.Position, aimPos)
            Camera.CFrame = Camera.CFrame:Lerp(desiredCF, 0.05)
            return
        end
        local maxError = 30
        local factor = math.min(angularError / maxError, 1) * 0.2
        local easedFactor = getEasedValue(factor, "Quad", "Out")
        if not shouldAssist then easedFactor = 0.05 end
        local desiredCF = CFrame.lookAt(Camera.CFrame.Position, aimPos)
        Camera.CFrame = Camera.CFrame:Lerp(desiredCF, easedFactor)
    else
        local desiredCF = CFrame.lookAt(Camera.CFrame.Position, aimPos)
        if getgenv().SmoothnessEnabled then
            local factor = getgenv().SmoothnessX or 0.1
            factor = math.min(factor, 1)
            Camera.CFrame = Camera.CFrame:Lerp(desiredCF, factor)
        else
            Camera.CFrame = desiredCF
        end
    end
end)
