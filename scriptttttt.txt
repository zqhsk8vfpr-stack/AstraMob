local function getCurrentPing()
    local stats = game:GetService("Stats")
    local network = stats:FindFirstChild("Network")
    if network then
        local ping = network:FindFirstChild("Ping")
        if ping and ping:IsA("NumberValue") then
            return ping.Value
        end
    end
    return nil
end

local function getPingBucket(ping)
    if not ping then return 60 end
    if ping < 65 then return 60
    elseif ping < 75 then return 70
    elseif ping < 85 then return 80
    elseif ping < 95 then return 90
    else return 100 end
end

local function IsKnocked(char)
    if not char then return true end
    local root = char:FindFirstChild("HumanoidRootPart")
    if root and root.Anchored then return true end
    for _, v in ipairs(char:GetDescendants()) do
        if v:IsA("Motor6D") and not v.Enabled then return true end
    end
    local hum = char:FindFirstChildOfClass("Humanoid")
    return hum and hum.Health <= 0
end

local function isVisible(targetPos, targetPlayer)
    local origin = Camera.CFrame.Position
    local direction = targetPos - origin
    if direction.Magnitude < 0.001 then return false end
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Blacklist
    local ignoreList = {}
    if LocalPlayer.Character then table.insert(ignoreList, LocalPlayer.Character) end
    if targetPlayer and targetPlayer.Character then table.insert(ignoreList, targetPlayer.Character) end
    rayParams.FilterDescendantsInstances = ignoreList
    local hit = Workspace:Raycast(origin, direction, rayParams)
    return hit == nil
end

local function hasForcefield(char)
    return char and char:FindFirstChildOfClass("ForceField") ~= nil
end

local function isFriend(plr)
    if getgenv().AimbotFriendCheck or getgenv().FriendCheck or getgenv().SilentAimFriendCheck then
        return plr:IsFriendsWith(LocalPlayer.UserId)
    end
    return false
end

local function isValidEnemy(plr)
    if plr == LocalPlayer then return false end
    if isFriend(plr) then return false end
    local char = plr.Character
    if not char then return false end
    if getgenv().AimbotKnockCheck and IsKnocked(char) then return false end
    if getgenv().AimbotForcefieldCheck and hasForcefield(char) then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then
        root = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or char:FindFirstChild("LowerTorso")
        if not root then return false end
    end
    return true
end

local function getAimPart(char, targetPlayer)
    if not char then return nil, nil end
    local mode = getgenv().HitpartSelection or "Direct"
    if mode == "Direct" then
        local head = char:FindFirstChild("Head")
        if head and head:IsA("BasePart") then return head, nil end
        local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
        if torso and torso:IsA("BasePart") then return torso, nil end
        return char:FindFirstChild("HumanoidRootPart"), nil
    end
    if mode == "Closest" then
        local crosshair = Camera.ViewportSize / 2
        local allParts = {}
        for _, child in ipairs(char:GetDescendants()) do
            if child:IsA("BasePart") and child.Transparency < 0.5 then
                local name = child.Name:lower()
                if not name:find("handle") and not name:find("tool") and not name:find("attachment") and not name:find("accessory") then
                    table.insert(allParts, child)
                end
            end
        end
        table.sort(allParts, function(a, b)
            local posA, onA = Camera:WorldToViewportPoint(a.Position)
            local posB, onB = Camera:WorldToViewportPoint(b.Position)
            if not onA and not onB then return false end
            if not onA then return false end
            if not onB then return true end
            local distA = (Vector2.new(posA.X, posA.Y) - crosshair).Magnitude
            local distB = (Vector2.new(posB.X, posB.Y) - crosshair).Magnitude
            return distA < distB
        end)
        for _, part in ipairs(allParts) do
            if getgenv().AimbotWallCheck and not isVisible(part.Position, targetPlayer) then continue end
            return part, nil
        end
        return allParts[1] or char:FindFirstChild("HumanoidRootPart"), nil
    end
    if mode == "Nearest" then
        local crosshair = Camera.ViewportSize / 2
        local ray = Camera:ViewportPointToRay(crosshair.X, crosshair.Y)
        local rayOrigin = ray.Origin
        local rayDirection = ray.Direction * 10000
        local rayParams = RaycastParams.new()
        rayParams.FilterType = Enum.RaycastFilterType.Whitelist
        local whitelist = {}
        for _, child in ipairs(char:GetDescendants()) do
            if child:IsA("BasePart") and child.Transparency < 0.5 then
                local name = child.Name:lower()
                if not name:find("handle") and not name:find("tool") and not name:find("attachment") and not name:find("accessory") then
                    table.insert(whitelist, child)
                end
            end
        end
        if #whitelist == 0 then return char:FindFirstChild("HumanoidRootPart"), nil end
        rayParams.FilterDescendantsInstances = whitelist
        local result = Workspace:Raycast(rayOrigin, rayDirection, rayParams)
        if result then
            if getgenv().AimbotWallCheck and not isVisible(result.Position, targetPlayer) then
                for _, part in ipairs(whitelist) do
                    if not getgenv().AimbotWallCheck or isVisible(part.Position, targetPlayer) then
                        return part, part.Position
                    end
                end
            else
                return result.Instance, result.Position
            end
        end
        local allParts = {}
        for _, child in ipairs(char:GetDescendants()) do
            if child:IsA("BasePart") and child.Transparency < 0.5 then
                local name = child.Name:lower()
                if not name:find("handle") and not name:find("tool") and not name:find("attachment") and not name:find("accessory") then
                    table.insert(allParts, child)
                end
            end
        end
        table.sort(allParts, function(a, b)
            local posA, onA = Camera:WorldToViewportPoint(a.Position)
            local posB, onB = Camera:WorldToViewportPoint(b.Position)
            if not onA and not onB then return false end
            if not onA then return false end
            if not onB then return true end
            local distA = (Vector2.new(posA.X, posA.Y) - crosshair).Magnitude
            local distB = (Vector2.new(posB.X, posB.Y) - crosshair).Magnitude
            return distA < distB
        end)
        for _, part in ipairs(allParts) do
            if getgenv().AimbotWallCheck and not isVisible(part.Position, targetPlayer) then continue end
            return part, nil
        end
        return allParts[1] or char:FindFirstChild("HumanoidRootPart"), nil
    end
    return char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart"), nil
end

local function selectTarget(enemies)
    if #enemies == 0 then return nil end
    local priority = getgenv().TargetPriority or "Closest"
    if priority == "Enemy look direction" then
        local bestDot, best = -2, nil
        for _, data in ipairs(enemies) do
            local root = data.rootPart
            if root then
                local dir = (root.Position - Camera.CFrame.Position).Unit
                local dot = Camera.CFrame.LookVector:Dot(dir)
                if dot > bestDot then
                    bestDot = dot
                    best = data
                end
            end
        end
        return best
    else
        local screenCenter = Camera.ViewportSize / 2
        local best, bestDist = nil, math.huge
        for _, data in ipairs(enemies) do
            local root = data.rootPart
            if not root then continue end
            local pos, onScreen = Camera:WorldToViewportPoint(root.Position)
            local dist
            if onScreen then
                dist = (Vector2.new(pos.X, pos.Y) - screenCenter).Magnitude
            else
                dist = (root.Position - Camera.CFrame.Position).Magnitude * 2
            end
            if dist < bestDist then
                bestDist = dist
                best = data
            end
        end
        return best
    end
end

local function IsAirborne(char)
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum and hum:GetState() == Enum.HumanoidStateType.Freefall then return true end
    local root = char:FindFirstChild("HumanoidRootPart")
    if root then
        local velY = root.AssemblyLinearVelocity.Y
        if velY > 2 or velY < -2 then return true end
    end
    return false
end

local function GetGroundIntersection(aimPos, targetChar)
    local origin = Camera.CFrame.Position
    local direction = aimPos - origin
    if direction.Magnitude < 0.001 then return nil end
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Blacklist
    rayParams.FilterDescendantsInstances = {LocalPlayer.Character, targetChar}
    local result = Workspace:Raycast(origin, direction, rayParams)
    if result then
        local hitPart = result.Instance
        if hitPart and hitPart:IsA("BasePart") then
            return result.Position
        end
    end
    return nil
end

local function getBodyParts(char)
    local parts = {}
    if not char then return parts end
    for _, child in ipairs(char:GetDescendants()) do
        if child:IsA("BasePart") and child.Transparency < 0.5 then
            local name = child.Name:lower()
            if not name:find("handle") and not name:find("tool") and not name:find("attachment") and not name:find("accessory") then
                table.insert(parts, child)
            end
        end
    end
    return parts
end

local function getValidEnemies()
    local enemies = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if isValidEnemy(plr) then
            local char = plr.Character
            local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or char:FindFirstChild("LowerTorso")
            table.insert(enemies, {
                player = plr,
                character = char,
                humanoid = char:FindFirstChildOfClass("Humanoid"),
                rootPart = root
            })
        end
    end
    return enemies
end

local function getKeyCode(name)
    if typeof(name) == "EnumItem" then return name end
    if type(name) ~= "string" then name = tostring(name) end
    for _, code in pairs(Enum.KeyCode:GetEnumItems()) do
        if code.Name == name then return code end
    end
    return Enum.KeyCode.E
end

local function getSilentAimPart(char, targetPlayer)
    if not char then return nil, nil end
    local method = getgenv().SilentAimMethod or "Direct"
    if method == "Direct" then
        local part = char:FindFirstChild("Head") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or char:FindFirstChild("HumanoidRootPart")
        return part, nil
    end
    if method == "Closest Part" then
        local crosshair = Camera.ViewportSize / 2
        local allParts = getBodyParts(char)
        table.sort(allParts, function(a, b)
            local posA, onA = Camera:WorldToViewportPoint(a.Position)
            local posB, onB = Camera:WorldToViewportPoint(b.Position)
            if not onA and not onB then return false end
            if not onA then return false end
            if not onB then return true end
            local distA = (Vector2.new(posA.X, posA.Y) - crosshair).Magnitude
            local distB = (Vector2.new(posB.X, posB.Y) - crosshair).Magnitude
            return distA < distB
        end)
        for _, part in ipairs(allParts) do
            if getgenv().SilentAimWallCheck and not isVisible(part.Position, targetPlayer) then continue end
            return part, nil
        end
        return allParts[1] or char:FindFirstChild("HumanoidRootPart"), nil
    end
    if method == "Nearest Point" then
        local crosshair = Camera.ViewportSize / 2
        local ray = Camera:ViewportPointToRay(crosshair.X, crosshair.Y)
        local rayOrigin = ray.Origin
        local rayDirection = ray.Direction * 10000
        local rayParams = RaycastParams.new()
        rayParams.FilterType = Enum.RaycastFilterType.Whitelist
        local whitelist = getBodyParts(char)
        if #whitelist == 0 then return char:FindFirstChild("HumanoidRootPart"), nil end
        rayParams.FilterDescendantsInstances = whitelist
        local result = Workspace:Raycast(rayOrigin, rayDirection, rayParams)
        if result then
            if getgenv().SilentAimWallCheck and not isVisible(result.Position, targetPlayer) then
                for _, part in ipairs(whitelist) do
                    if not getgenv().SilentAimWallCheck or isVisible(part.Position, targetPlayer) then
                        return part, part.Position
                    end
                end
            else
                return result.Instance, result.Position
            end
        end
        local allParts = getBodyParts(char)
        table.sort(allParts, function(a, b)
            local posA, onA = Camera:WorldToViewportPoint(a.Position)
            local posB, onB = Camera:WorldToViewportPoint(b.Position)
            if not onA and not onB then return false end
            if not onA then return false end
            if not onB then return true end
            local distA = (Vector2.new(posA.X, posA.Y) - crosshair).Magnitude
            local distB = (Vector2.new(posB.X, posB.Y) - crosshair).Magnitude
            return distA < distB
        end)
        for _, part in ipairs(allParts) do
            if getgenv().SilentAimWallCheck and not isVisible(part.Position, targetPlayer) then continue end
            return part, part.Position
        end
        return char:FindFirstChild("HumanoidRootPart"), nil
    end
    return char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart"), nil
end

local function computeSilentAimPosition(part, targetPlayer, usePrediction, useAutoPred, accuracy)
    if not part then return nil end
    local aimPos = part.Position
    local vel = part.AssemblyLinearVelocity
    if usePrediction then
        local hPred = getgenv().SilentAimPredX or 0.115
        local vPred = getgenv().SilentAimPredY or 0.115
        local zPred = getgenv().SilentAimPredZ or 0.115
        aimPos = aimPos + Vector3.new(vel.X * hPred, vel.Y * vPred, vel.Z * zPred)
    end
    if useAutoPred then
        local hMult = getgenv().SilentAimPredX or 0.115
        local vMult = getgenv().SilentAimPredY or 0.115
        local zMult = getgenv().SilentAimPredZ or 0.115
        aimPos = aimPos + Vector3.new(vel.X * hMult, vel.Y * vMult, vel.Z * zMult)
    end
    if accuracy and accuracy < 100 then
        local spread = (100 - accuracy) / 100 * 0.04
        local angle = math.random() * 2 * math.pi
        local randomDir = Vector3.new(math.cos(angle), math.sin(angle), 0) * spread
        aimPos = aimPos + randomDir
    end
    return aimPos
end

local function prepareSilentAim()
    print("[Silent] prepareSilentAim called")
    if not getgenv().SilentAimEnabled then 
        print("[Silent] silent aim disabled") 
        return 
    end
    local hitChance = getgenv().SilentAimHitChance or 85
    if math.random(100) > hitChance then 
        print("[Silent] hit chance failed") 
        return 
    end

    local targetPlayer = SilentTarget
    if not targetPlayer or not targetPlayer.character then 
        print("[Silent] SilentTarget is nil or no character") 
        return 
    end
    print("[Silent] target:", targetPlayer.player.Name)

    local targetChar = targetPlayer.character
    if getgenv().SilentAimKnockCheck and IsKnocked(targetChar) then 
        print("[Silent] knocked") 
        return 
    end
    if getgenv().SilentAimForcefieldCheck and hasForcefield(targetChar) then 
        print("[Silent] forcefield") 
        return 
    end
    if getgenv().SilentAimFriendCheck and targetPlayer.player:IsFriendsWith(LocalPlayer.UserId) then 
        print("[Silent] friend") 
        return 
    end

    local hum = targetChar:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then 
        print("[Silent] dead") 
        return 
    end

    local aimPart, customPoint = getSilentAimPart(targetChar, targetPlayer)
    if not aimPart then 
        print("[Silent] no aim part") 
        return 
    end

    if getgenv().SilentAimWallCheck and not isVisible(aimPart.Position, targetPlayer) then 
        print("[Silent] wall check") 
        return 
    end

    local aimPos = customPoint or computeSilentAimPosition(aimPart, targetPlayer,
        getgenv().SilentAimPrediction,
        getgenv().SilentAimAutoPrediction,
        getgenv().SilentAimHitChance or 85)

    local antiCurve = getgenv().SilentAimAntiCurve or 8
    if antiCurve > 0 then
        local dir = (aimPos - Camera.CFrame.Position).Unit
        local angle = math.deg(math.acos(Camera.CFrame.LookVector:Dot(dir)))
        if angle > antiCurve then 
            print("[Silent] anti-curve, angle:", angle) 
            return 
        end
    end

    if getgenv().SilentAimAntiGround then
        local groundHit = GetGroundIntersection(aimPos, targetChar)
        if groundHit then 
            print("[Silent] anti-ground") 
            return 
        end
    end

    print("[Silent] aimPos computed:", aimPos)
    fireSilentAim(aimPos)
end

local prevLookVectors = {}
local function recordLookVector()
    table.insert(prevLookVectors, Camera.CFrame.LookVector)
    if #prevLookVectors > 5 then
        table.remove(prevLookVectors, 1)
    end
end

local function getMovementDirection()
    if #prevLookVectors < 2 then return nil end
    local avg = Vector3.zero
    for i = 2, #prevLookVectors do
        avg = avg + (prevLookVectors[i] - prevLookVectors[i-1])
    end
    return avg.Unit
end

local function getEasedValue(t, style, dir)
    style = style or "Linear"
    dir = dir or "Out"
    if style == "Quad" and dir == "Out" then
        return 1 - (1 - t) * (1 - t)
    end
    return t
end

RunService.RenderStepped:Connect(function()
    recordLookVector()

    if not getgenv().Enabled then
        currentTarget = nil
        SilentTarget = nil
        return
    end

    local activation = getgenv().Activation or "Hold"
    local shouldLock = false
    local enemiesList = nil

    if activation == "Rage" then
        shouldLock = true
        enemiesList = getValidEnemies()
    elseif activation == "Mobile" then
        if isTouching then
            shouldLock = true
            enemiesList = getValidEnemies()
        else
            currentTarget = nil
            SilentTarget = nil
            return
        end
    elseif activation == "Hold" then
        local key = getKeyCode(getgenv().AimbotKey or "E")
        if UserInputService:IsKeyDown(key) then
            shouldLock = true
            enemiesList = getValidEnemies()
        else
            currentTarget = nil
            SilentTarget = nil
            return
        end
    elseif activation == "Toggle" then
        if camLockActive then
            shouldLock = true
            enemiesList = getValidEnemies()
        else
            currentTarget = nil
            SilentTarget = nil
            return
        end
    end

    if not shouldLock then
        currentTarget = nil
        SilentTarget = nil
        return
    end

    if not enemiesList or #enemiesList == 0 then
        currentTarget = nil
        SilentTarget = nil
        return
    end

    if currentTarget then
        local stillValid = false
        for _, data in ipairs(enemiesList) do
            if data.player == currentTarget.player then
                stillValid = true
                break
            end
        end
        if not stillValid then
            currentTarget = nil
            SilentTarget = nil
        end
    end

    if not currentTarget then
        local targetData = selectTarget(enemiesList)
        if targetData then
            currentTarget = targetData
        else
            currentTarget = enemiesList[1]
        end
        if not currentTarget then
            return
        end
    end

    if getgenv().SilentAimEnabled then
        SilentTarget = currentTarget
    else
        SilentTarget = nil
    end

    local char = currentTarget.character
    local root = currentTarget.rootPart

    local aimPart, customPoint = getAimPart(char, currentTarget.player)
    if not aimPart then
        aimPart = root
    end

    local aimPos = customPoint or aimPart.Position

    if getgenv().Prediction then
        local vel = aimPart.AssemblyLinearVelocity
        local hPred = getgenv().PredictionX or 0.115
        local vPred = getgenv().PredictionY or 0.115
        local zPred = getgenv().PredictionZ or 0.115
        aimPos = aimPos + Vector3.new(vel.X * hPred, vel.Y * vPred, vel.Z * zPred)
    end

    if getgenv().AutoPrediction then
        local hMult = getgenv().PredictionX or 0.115
        local vMult = getgenv().PredictionY or 0.115
        local zMult = getgenv().PredictionZ or 0.115
        local vel = aimPart.AssemblyLinearVelocity
        aimPos = aimPos + Vector3.new(vel.X * hMult, vel.Y * vMult, vel.Z * zMult)
    end

    local airborne = IsAirborne(char)
    if airborne then
        local jumpOff = getgenv().JumpOffset or 0
        if jumpOff ~= 0 then
            aimPos = aimPos + Vector3.new(0, jumpOff, 0)
        end
    else
        local fallOff = getgenv().FallOffset or 0
        if fallOff ~= 0 then
            aimPos = aimPos + Vector3.new(0, fallOff, 0)
        end
    end

    if getgenv().AimbotWallCheck and not isVisible(aimPos, currentTarget.player) then
        return
    end

    if getgenv().TargetOutline then
        local dirToTarget = (aimPos - Camera.CFrame.Position).Unit
        local angularError = math.deg(math.acos(Camera.CFrame.LookVector:Dot(dirToTarget)))
        local ray = Camera:ViewportPointToRay(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
        local rel = aimPos - ray.Origin
        local proj = rel:Dot(ray.Direction)
        local closestOnRay = ray.Origin + ray.Direction * proj
        local perpDist = (aimPos - closestOnRay).Magnitude

        local shouldAssist = true
        if perpDist > 8 then
            shouldAssist = false
        end
        if angularError < 2.0 then
            local desiredCF = CFrame.lookAt(Camera.CFrame.Position, aimPos)
            Camera.CFrame = Camera.CFrame:Lerp(desiredCF, 0.05)
            return
        end

        local maxError = 30
        local factor = math.min(angularError / maxError, 1) * 0.2
        local easedFactor = getEasedValue(factor, "Quad", "Out")
        if not shouldAssist then
            easedFactor = 0.05
        end

        local desiredCF = CFrame.lookAt(Camera.CFrame.Position, aimPos)
        Camera.CFrame = Camera.CFrame:Lerp(desiredCF, easedFactor)
    else
        local desiredCF = CFrame.lookAt(Camera.CFrame.Position, aimPos)
        if getgenv().SmoothnessEnabled then
            local factor = getgenv().SmoothnessX or 0.1
            factor = math.min(factor, 1)
            Camera.CFrame = Camera.CFrame:Lerp(desiredCF, factor)
        else
            Camera.CFrame = desiredCF
        end
    end
end)

                                    pcall(function() reloadRemote:FireServer() end)

                                        if firekey then
                                            firekey(rKey, "down")
                                            task.wait(0.05)

                                    end)
                                    lastAutoReloadTime = now

local lastBuyTime = 0
local buyableCache = {}
local cacheTime = 0

local function tryBuyItem(clickable)
    if not clickable then return false end
    if clickable:IsA("ClickDetector") then
        pcall(fireclickdetector, clickable)
        return true
    elseif clickable:IsA("ProximityPrompt") and clickable.Enabled then
        pcall(function() clickable:Prompt() end)
        return true
    end
    return false
end

local function cleanItemName(name)
    if not name then return "" end
    return name:gsub("[%[%]]", ""):gsub("^%s*(.-)%s*$", "%1"):lower()
end

local function getItemNameFromObject(obj)
    if not obj then return nil end
    local parent = obj.Parent
    if parent then
        local pname = parent.Name
        if pname and pname ~= "" and pname ~= "ClickDetector" and pname ~= "ProximityPrompt" and pname ~= "Part" and pname ~= "BasePart" then
            return pname
        end
        local grandParent = parent.Parent
        if grandParent then
            local gpname = grandParent.Name
            if gpname and gpname ~= "" and gpname ~= "Workspace" and gpname ~= "Model" and gpname ~= "Folder" then
                return gpname
            end
        end
    end
    local searchRoot = parent or obj
    for _, child in ipairs(searchRoot:GetDescendants()) do
        if child:IsA("StringValue") and child.Name:lower():find("name") then
            return child.Value
        elseif child:IsA("TextLabel") or child:IsA("TextButton") then
            if child.Text and child.Text ~= "" then
                return child.Text
            end
        elseif child:IsA("Hint") then
            if child.Text and child.Text ~= "" then
                return child.Text
            end
        end
    end
    local model = obj:FindFirstAncestorOfClass("Model")
    if model and model.Name and model.Name ~= "" then
        return model.Name
    end
    return nil
end

local function refreshBuyableCache()
    buyableCache = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("ClickDetector") or obj:IsA("ProximityPrompt") then
            local parent = obj.Parent
            local part = nil
            if parent then
                part = parent:FindFirstChildWhichIsA("BasePart")
                if not part then
                    for _, child in ipairs(parent:GetDescendants()) do
                        if child:IsA("BasePart") then
                            part = child
                            break
                        end
                    end
                end
            end
            if not part then
                if obj:IsA("BasePart") then part = obj end
            end
            if part then
                local name = getItemNameFromObject(obj) or "unknown"
                local cleaned = cleanItemName(name)
                table.insert(buyableCache, {
                    clickable = obj,
                    part = part,
                    name = name,
                    cleaned = cleaned:lower()
                })
            end
        end
    end
    cacheTime = tick()
end

task.spawn(function()
    while true do
        task.wait(0.5)
        pcall(function()
            local buyArmor = getgenv().AutoBuyArmor or false
            local buyFood = getgenv().AutoBuyFood or false
            local buyWeapons = getgenv().AutoBuyWeapons or false
            if not buyArmor and not buyFood and not buyWeapons then return end

            local now = tick()
            local cooldown = getgenv().AutoBuyCooldown or 5
            if now - lastBuyTime < cooldown then return end

            local char = LocalPlayer.Character
            if not char then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then return end

            local proximity = getgenv().AutoBuyProximity or 20
            local foodList = getgenv().AutoBuyFoodList or {}
            local weaponList = getgenv().AutoBuyWeaponsList or {}

            local cleanFoodList = {}
            for _, item in ipairs(foodList) do
                local cleaned = cleanItemName(item):lower()
                if cleaned ~= "" then table.insert(cleanFoodList, cleaned) end
            end
            local cleanWeaponList = {}
            for _, item in ipairs(weaponList) do
                local cleaned = cleanItemName(item):lower()
                if cleaned ~= "" then table.insert(cleanWeaponList, cleaned) end
            end

            if #buyableCache == 0 or (now - cacheTime) > 2 then
                refreshBuyableCache()
            end

            for _, item in ipairs(buyableCache) do
                local distance = (item.part.Position - hrp.Position).Magnitude
                if distance > proximity then continue end

                local cleanName = item.cleaned
                if cleanName == "" then cleanName = item.name:lower() end

                local shouldBuy = false
                if buyArmor and (cleanName:find("armor") or cleanName:find("medium") or cleanName:find("high")) then
                    shouldBuy = true
                end

                if not shouldBuy and buyFood then
                    for _, food in ipairs(cleanFoodList) do
                        if cleanName:find(food) then
                            shouldBuy = true
                            break
                        end
                    end
                end

                if not shouldBuy and buyWeapons then
                    for _, weap in ipairs(cleanWeaponList) do
                        if cleanName:find(weap) then
                            shouldBuy = true
                            break
                        end
                    end
                end

                if shouldBuy then
                    if tryBuyItem(item.clickable) then
                        lastBuyTime = now
                        return
                    end
                end
            end
        end)
    end
end)
